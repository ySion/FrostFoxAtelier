cmake_minimum_required(VERSION 4.2)
project(FrostFoxLayoutStep1Probe LANGUAGES C CXX)
# Native research executable only; this is not the production layout library.
get_filename_component(ATELIER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${ATELIER_ROOT}/cmake/mimalloc.cmake")
add_library(layout_step1_kernel STATIC kernel.cpp)
target_include_directories(layout_step1_kernel PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(layout_step1_probe main.cpp "${ATELIER_ROOT}/src/memory/global_new_delete.cpp")
foreach(target layout_step1_kernel layout_step1_probe)
  set_target_properties(${target} PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  target_compile_options(${target} PRIVATE /W4 /WX /EHsc /permissive- /utf-8)
endforeach()
set_target_properties(layout_step1_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_compile_definitions(layout_step1_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(layout_step1_probe PRIVATE layout_step1_kernel mimalloc dwrite)
target_link_options(layout_step1_probe PRIVATE /include:mi_version)
